%% NBR 10520:2023 citation conformance test.
%% Each example is prefixed with @@key@@ so the runner can compare citations
%% individually.  The tests exercise author-date citations; note styles are
%% covered by tests/abnt-ibid.

\usepackage[paperwidth=120cm,paperheight=120cm,margin=1cm]{geometry}
\usepackage[brazil]{babel}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{lmodern}
\usepackage[style=abnt]{biblatex}
\usepackage[autostyle]{csquotes}

\addbibresource{NBR10520-2023.bib}
\pagestyle{empty}
\setlength{\parindent}{0pt}
